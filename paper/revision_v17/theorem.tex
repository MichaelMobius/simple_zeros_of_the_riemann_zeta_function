\begin{theorem}[Main theorem]\label{thm:main}
With
\[
 H_{\rm MT}
 =
 \frac32-\frac1{\sqrt2}\cot\frac1{\sqrt2},
\]
one has unconditionally
\[
 \liminf_{T\to\infty}
 \frac{N_0^s(T,2T)}{N(T,2T)}
 \ge
 \frac{1125000H_{\rm MT}-2220}{1120671}
 =
 0.6731175265883904388095857434106058666\ldots .
\]
\end{theorem}
